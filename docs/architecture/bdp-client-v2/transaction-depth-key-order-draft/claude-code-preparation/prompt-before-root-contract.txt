export function transactionInputText(value: LosslessJsonValue): string {
  const pending: (string | { value: LosslessJsonValue })[] = [{ value }];
  const chunks: string[] = [];
  while (pending.length) {
    const task = pending.pop()!;
    if (typeof task === "string") {
      chunks.push(task);
      continue;
    }
    const item = task.value;
    if (item instanceof JsonNumberLiteral) chunks.push(item.literal);
    else if (Array.isArray(item)) {
      chunks.push("[");
      pending.push("]");
      for (let i = item.length - 1; i >= 0; i--) {
        pending.push({ value: item[i]! });
        if (i > 0) pending.push(",");
      }
    } else if (item && typeof item === "object") {
      chunks.push("{");
      pending.push("}");
      const entries = Object.entries(item);
      for (let i = entries.length - 1; i >= 0; i--) {
        const [key, child] = entries[i]!;
        pending.push({ value: child }, ":", JSON.stringify(key));
        if (i > 0) pending.push(",");
      }
    } else chunks.push(JSON.stringify(item));
  }
  return chunks.join("");
}

    controls: TransactionReceivingControls | undefined,

  function admitKey(
    principal: string,
    key: string,
    batch: PreparedTransaction,
    pendingWins = false,
    references?: ReferenceBindings,
    template?: DependencyTemplate,
    controls?: TransactionReceivingControls,
  ): TransactionReceivingAdmission {
    const stored = row(principal, key);
    const prior = stored ? expireRow(stored, instant()).row : undefined;
    if (pendingWins && prior?.owner !== null && prior !== undefined)
      return { kind: "existing", receipt: parseReceipt(prior.receipt) };
    if (prior?.dependency) {
      const template = parseDependencyTemplate(prior.dependency, scope);
      const comparison = compareDependencyTemplate(template, batch, resolveAlias, (index) => {
        const stored = db
          .prepare("SELECT fact FROM creator_facts WHERE attempt=? AND member=?")
          .get(template.attempt, index);
        if (!stored) throw new Error("unresolved receipt lost creator fact");
        return typeof stored.fact === "string"
          ? (JSON.parse(stored.fact) as CreatorFact)
          : undefined;
      });
      return comparison === "conflict"
        ? { kind: "conflict" }
        : comparison === "equal"
          ? { kind: "existing", receipt: parseReceipt(prior.receipt) }
          : { kind: "comparison-pending" };
    }
    const depthRefusal = transactionDepthRefusal(batch);
    if (!prior && depthRefusal !== undefined) {
      if (controls === undefined) throw new TransactionDevelopmentLimitError(depthRefusal);
      return { kind: "development-limit", message: depthRefusal };
    }
    const normalized = template
      ? undefined
      : normalizeTransaction(batch, resolveAlias, prior?.witnesses, references);
    if (prior) {
      if (!normalized) throw new Error("existing key cannot acquire a new unresolved template");
      return prior.identity !== normalized.identity
        ? { kind: "conflict" }
        : { kind: "existing", receipt: parseReceipt(prior.receipt) };
    }
    // Prior-key equality/conflict wins above. Only an unknown key pays fresh
    // admission controls, and no refusal ever inserts a receipt or owner.
    if (controls && batch.carrier.operations.length > controls.maximumOperations)
      return { kind: "operation-limit" };
    if (controls) {
      for (const [index, op] of batch.carrier.operations.entries()) {
        if (op.operation !== "updateWhere" && op.operation !== "deleteWhere") continue;
        const limit = transactionSelectorCapacity(op);
        if (limit)
          return {
            kind: "selector-limit",
            limit,
            pointer:
              batch.carrier.kind === "singleton" ? "/selector" : `/operations/${index}/selector`,
          };
      }
    }
    if (controls && !controls.acceptsJson) return { kind: "not-acceptable" };
    const receipt: TransactionReceipt = {
      id: `${scope}receipts/${randomUUID()}`,
      status: "pending",
      idempotencyKey: key,
      scopeEpoch: epoch,
      authorizationView: view,
      transaction: randomUUID(),
    };
    const owner = randomUUID(),
      claim = Object.freeze({ receipt });
    db.prepare("INSERT INTO receipts VALUES(?,?,?,?,?,?,?)").run(
      principal,
      key,
      normalized?.identity ?? "",
      normalized?.witnessesJson ?? "",
      owner,
      json(receipt),
      template ? json(template) : null,
    );
    claims.set(claim, { principal, key, owner, normalized });
    return { kind: "owned", claim };
  }

import type { PreparedTransaction } from "./transaction-batch.js";
import {
  snapshotTransactionReceivingControls,
  type openTransactionStore,
  type TransactionReceivingAdmission,
  type TransactionReceivingControls,
  type TransactionReadContext,
} from "./transaction-store.js";
export type DirectComparison =
  | Exclude<TransactionReceivingAdmission, { kind: "comparison-pending" }>
  | { readonly kind: "temporarily-unavailable" }
  | { readonly kind: "comparison-deadline"; readonly context: TransactionReadContext }
  | { readonly kind: "comparison-owner-stopped" };
/** The private default is finite and independent of transaction duration. Options
 * are test seams, not protocol/discovery configuration or another key owner. */
export async function compareAndAdmitTransaction(
  store: Pick<ReturnType<typeof openTransactionStore>, "scopeEpoch" | "comparisonEpoch"> & {
    admit(
      principal: string,
      key: string,
      request: PreparedTransaction,
      controls?: TransactionReceivingControls,
    ): TransactionReceivingAdmission;
  },
  principal: string,
  key: string,
  request: PreparedTransaction,
  options: {
    readonly receiving?: TransactionReceivingControls;
    readonly budgetMs?: number;
    readonly now?: () => number;
    readonly wait?: () => Promise<void>;
    /** Bound by observed submission only; never runtime/caller configuration. */
    readonly observeDeadline?:
      | ReturnType<typeof openTransactionStore>["observeComparisonDeadline"]
      | undefined;
  } = {},
): Promise<DirectComparison> {
  const receiving =
    options.receiving === undefined
      ? undefined
      : snapshotTransactionReceivingControls(options.receiving);
  const budget = options.budgetMs ?? 1000;
  if (!Number.isSafeInteger(budget) || budget <= 0 || budget > 2147483647)
    throw new TypeError("finite positive comparison budget required");
  const now = options.now ?? (() => performance.now());
  const deadline = now() + budget;
  const epoch = store.scopeEpoch;
  for (;;) {
    if (store.comparisonEpoch() !== epoch)
      return options.observeDeadline
        ? { kind: "comparison-owner-stopped" }
        : { kind: "temporarily-unavailable" };
    if (now() >= deadline) {
      if (!options.observeDeadline) return { kind: "temporarily-unavailable" };
      // Synchronous final check + current tuple; never hold its owner entry over wait.
      const observed = options.observeDeadline(epoch);
      return observed.kind === "observed"
        ? { kind: "comparison-deadline", context: observed.context }
        : { kind: "comparison-owner-stopped" };
    }
    // Retraction may make this a fresh admission. Same original deadline and no
    // inherited claim: admit consults the current authoritative key each time.
    const result =
      receiving === undefined
        ? store.admit(principal, key, request)
        : store.admit(principal, key, request, receiving);
    if (result.kind !== "comparison-pending") return result;
    await (options.wait ? options.wait() : delay(Math.min(5, Math.max(0, deadline - now()))));
  }

    prepared = prepareTransactionForReceiving(store.scope, submission.carrier, body);

    return { kind: "refused", status: 400, code: "malformed-request" };

    return { kind: "refused", status: 401, code: "unauthenticated" };

    return { kind: "refused", status: 403, code: "forbidden" };

      observeDeadline: observed ? store.observeComparisonDeadline : undefined,
    },

      result.kind === "operation-limit" ||
      result.kind === "selector-limit" ||

    throw new Error("admission refusal lost same-entry context");

/** Bounded inherited-IPC launcher. No request or observation emulation. */
import { spawn } from "node:child_process";
import { mkdir } from "node:fs/promises";
export async function launchInstalledTransaction(entry, configuration, signal) {
  signal?.throwIfAborted();
  // Explicit launcher-owned provisioning; reopen keeps the directory/create:false.
  if (configuration.create) await mkdir(configuration.directory, { recursive: true, mode: 0o700 });
  signal?.throwIfAborted();
  const child = spawn(process.execPath, [entry], {
    cwd: "/tmp",
    stdio: ["ignore", "pipe", "pipe", "ipc"],
  });
  let bytes = 0;
  for (const stream of [child.stdout, child.stderr])
    stream.on("data", (chunk) => {
      bytes += chunk.length;
      if (bytes > 65536) child.kill("SIGKILL");
    });
  const exited = new Promise((resolve) =>
    child.once("exit", (code, signal) => resolve({ code, signal })),
  );
  async function bounded(promise) {
    let timer;
    try {
      return await Promise.race([
        promise,
        new Promise((_, reject) => {
          timer = setTimeout(() => {
            child.kill("SIGKILL");
            reject(new Error("installed transaction process deadline"));
          }, 10000);
        }),
      ]);
    } finally {
      clearTimeout(timer);
    }
  }
  async function drainFailure(error) {
    child.kill("SIGKILL");
    try {
      await bounded(exited);
    } catch (cleanup) {
      throw new AggregateError([error, cleanup], "installed transaction cleanup failed");
    }
    throw error;
  }
  let closing;
  const close = () =>
    (closing ??= (async () => {
      try {
        if (child.connected) child.send({ command: "close" });
        const result = await bounded(exited);
        if (result.code !== 0 || result.signal !== null)
          throw new Error("installed transaction child did not close cleanly");
      } catch (error) {
        await drainFailure(error);
      } finally {
        signal?.removeEventListener("abort", onAbort);
      }
    })());
  const onAbort = () => {
    void close().catch(() => {});
  };
  signal?.addEventListener("abort", onAbort, { once: true });
  if (signal?.aborted) onAbort();
  let detachReadyAbort = () => {};
  try {
    const ready = await bounded(
      new Promise((resolve, reject) => {
        const abort = () => reject(signal.reason);
        signal?.addEventListener("abort", abort, { once: true });
        detachReadyAbort = () => signal?.removeEventListener("abort", abort);
        if (signal?.aborted) {
          abort();
          return;
        }
        child.once("error", reject);
        child.once("message", resolve);
        void exited.then(() => reject(new Error("installed transaction exited before ready")));
        child.send({ configuration });
      }),
    );
    signal?.throwIfAborted();
    if (!ready || ready.event !== "qualification.ready" || JSON.stringify(ready).length > 65536)
      throw new Error("invalid installed ready message");
    return { child, ready, close };
  } catch (error) {
    try {
      await close();
    } catch (cleanup) {
      throw new AggregateError([error, cleanup], "installed transaction startup cleanup failed");
    }
    throw error;
  } finally {
    detachReadyAbort();
  }
}
