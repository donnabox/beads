787:       "exact HTTP request configuration is invalid",
788:       {},

879: function exchangeRawBytes(
880:   route: RawHttpDialRoute,
881:   requestHead: Uint8Array,
882:   method: HttpMethod,
883:   signal: AbortSignal,
884:   requestTimeoutMs: number,
885:   maximumHeaderBytes: number,
886:   maximumBodyBytes: number,
887:   requestBody?: Uint8Array,
888:   observeWrite?: (state: RequestWriteState) => void,
889: ): Promise<{ bytes: Buffer; writeState: RequestWriteState }> {
890:   return new Promise((resolve, reject) => {
891:     let socket: Socket | undefined;
892:     let settled = false;
893:     let earlyCleanup: ReturnType<typeof setImmediate> | undefined;
894:     let connected = false;
895:     let state: RequestWriteState = "not-started";
896:     let completedWrites = 0;
897:     const requiredWrites = requestBody === undefined ? 1 : 2;
898:     const wireLimit = Math.min(Number.MAX_SAFE_INTEGER, maximumBodyBytes + maximumHeaderBytes * 2);
899:     let buffered = Buffer.alloc(Math.min(8192, wireLimit));
900:     let total = 0;
901:     const raw = (): Buffer => buffered.subarray(0, total);
902:     const recordState = (next: RequestWriteState): void => {
903:       if (settled) return;
904:       state = next;
905:       observeWrite?.(next);
906:     };
907:     const finish = (error?: HttpTransportError): void => {
908:       if (settled) return;
909:       settled = true;
910:       clearTimeout(timer);
911:       if (earlyCleanup !== undefined) clearImmediate(earlyCleanup);
912:       signal.removeEventListener("abort", onAbort);
913:       if (socket !== undefined) {
914:         socket.removeAllListeners();
915:         socket.on("error", () => undefined);
916:         socket.destroy();
917:       }
918:       if (error !== undefined) reject(error);
919:       else resolve({ bytes: Buffer.from(raw()), writeState: state });
920:       buffered = Buffer.alloc(0);
921:     };
922:     const completeResponse = (): boolean => {
923:       try {
924:         return isFramedResponseComplete(raw(), method, maximumHeaderBytes, maximumBodyBytes);
925:       } catch {
926:         return false;
927:       }
928:     };
929:     const writeDone = (error?: Error | null): void => {
930:       if (settled) return;
931:       if (error !== undefined && error !== null) {
932:         finish(
933:           completeResponse()
934:             ? undefined
935:             : new HttpTransportError("disconnect", "HTTP request write disconnected"),
936:         );
937:         return;
938:       }
939:       completedWrites++;
940:       if (completedWrites === requiredWrites) recordState("complete");
941:     };
942:     const onAbort = (): void =>
943:       finish(new HttpTransportError("abort", "HTTP exchange was aborted"));
944:     const timer = setTimeout(
945:       () => finish(new HttpTransportError("timeout", "HTTP exchange timed out")),
946:       requestTimeoutMs,
947:     );
948:     if (signal.aborted) {
949:       onAbort();
950:       return;
951:     }
952:     signal.addEventListener("abort", onAbort, { once: true });
953:     try {
954:       socket =
955:         route.transport === "plain"
956:           ? connectPlain({ host: route.host, port: route.port })
957:           : connectTls({
958:               host: route.host,
959:               port: route.port,
960:               ...(route.servername === undefined ? {} : { servername: route.servername }),
961:               ...(route.ca === undefined ? {} : { ca: route.ca }),
962:             });
963:     } catch {
964:       finish(new HttpTransportError("configuration", "HTTP dial route was invalid"));
965:       return;
966:     }
967:     const peer = socket;
968:     peer.once(route.transport === "plain" ? "connect" : "secureConnect", () => {
969:       if (settled) return;
970:       connected = true;
971:       recordState("started-completion-unestablished");
972:       try {
973:         // Bounded owned buffers; two writes avoid copying a large entity again.
974:         peer.write(requestHead, writeDone);
975:         if (requestBody !== undefined) peer.write(requestBody, writeDone);
976:       } catch {
977:         finish(
978:           completeResponse()
979:             ? undefined
980:             : new HttpTransportError("disconnect", "HTTP request write disconnected"),
981:         );
982:       }
983:     });
984:     peer.on("data", (chunk: Buffer) => {
985:       if (settled) return;
986:       if (chunk.length > wireLimit - total) {
987:         finish(
988:           new HttpTransportError("body-limit", "HTTP response exceeded the configured body limit"),
989:         );
990:         return;
991:       }
992:       const next = total + chunk.length;
993:       if (next > buffered.length) {
994:         const grown = Buffer.alloc(Math.min(wireLimit, Math.max(next, buffered.length * 2)));
995:         buffered.copy(grown, 0, 0, total);
996:         buffered = grown;
997:       }
998:       chunk.copy(buffered, total);
999:       total = next;
1000:       try {
1001:         const head = locateFinalResponseHead(raw(), maximumHeaderBytes);
1002:         if (head === undefined) return;
1003:         if (total - head.headerEnd - 4 > maximumBodyBytes + maximumHeaderBytes)
1004:           throw new HttpTransportError(
1005:             "body-limit",
1006:             "HTTP response exceeded the configured body limit",
1007:           );
