export function prepareTransactionSequence(
  scope: string,
  text: string,
): PreparedTransactionSequence {
  const sequence = prepareSequence(scope, text);

  return sequence;
}

function prepareSequence(scope: string, text: string): PreparedTransactionSequence {
  if (Buffer.byteLength(text) > TRANSACTION_RUNTIME_LIMITS.requestBodyBytes)
    throw new RangeError("development sequence byte limit");
  const carrier = prepareReadUpdateSequence(scope, text);
  if (carrier.operations.length > TRANSACTION_MAXIMUM_OPERATIONS)
    throw new RangeError("development sequence member limit");
  const raw = object(decodeJsonDocument(text)).operations;
  if (!Array.isArray(raw)) throw new Error("parsed sequence lost operations");
  const labels = new Map<string, number>();
  const members = carrier.operations.map((op, index) => {
    const aliasOperands: { readonly field: string; readonly locator: string }[] = [];
    const dependencies: {
      field: string;
      creator: number;
      resourceKind: "bead" | "link";
      suppliedId?: string;
    }[] = [];
    for (const field of ["bead", "link", "source", "target"]) {
      const ref = op.input[field];
      if (ref === undefined) continue;
      const uri = typeof ref === "string" ? ref : (ref as { uri: string }).uri;
      if (uri.startsWith("@")) {
        const creator = labels.get(uri.slice(1));
        if (creator === undefined) throw new TypeError("unknown sequence creator");
        const creation = carrier.operations[creator];
        if (!creation) throw new Error("prepared sequence lost creator");
        const supplied = creation.input.id as string | undefined;
        dependencies.push({
          field,
          creator,
          resourceKind: creation.operation === "createBead" ? "bead" : "link",
          ...(supplied === undefined ? {} : { suppliedId: canonicalReference(scope, supplied) }),
        });
      }
      if (
        transactionAliasFields(op.operation).includes(field) &&
        canonicalReference(scope, uri).startsWith(`${scope}alias/`)
      )
        aliasOperands.push(Object.freeze({ field, locator: canonicalReference(scope, uri) }));
    }
    // The whole carrier already validated backward labels and Resource kinds.
    // Preserve its branded numeric operation without reparsing as a singleton,
    // whose syntax correctly rejects @name outside a sequence.
    const key = carrier.keys[index];
    if (key === undefined) throw new Error("prepared sequence lost member key");
    // Original raw member depth2 equals root0 / operations array1 / member2.
    // The complete carrier parser already validated every label and reference.
    const request = prepareDepthCarrier(
      Object.freeze({
        kind: "singleton",
        scope: carrier.scope,
        operations: Object.freeze([op]),
        keys: Object.freeze([key]),
      }),
      object(raw[index]),
      2,
    );
    if (op.name) labels.set(op.name, index);
    return Object.freeze({
      key,
      operation: op.operation,
      ...(op.name === undefined ? {} : { name: op.name }),
      request,
      dependencies: Object.freeze(dependencies.map((dependency) => Object.freeze(dependency))),
      aliasOperands: Object.freeze(aliasOperands),
      deferred: dependencies.length > 0 || aliasOperands.length > 0,
    });
  });
  const result = Object.freeze({ scope: carrier.scope, members: Object.freeze(members) });
  sequences.add(result);
  return result;
}
export function assertPreparedTransactionSequence(
  sequence: PreparedTransactionSequence,
  scope: string,
): void {
  if (!sequences.has(sequence) || sequence.scope !== scope)
    throw new TypeError("expected prepared transaction sequence for this Scope");
}
function prepareCarrier(carrier: PreparedTransaction["carrier"]): PreparedTransaction {
  const result = Object.freeze({ carrier });
  prepared.add(result);
  return result;
}
function prepareDepthCarrier(
  carrier: PreparedTransaction["carrier"],
  input: LosslessJsonValue,
  depth = 0,
): PreparedTransaction {
  const result = prepareCarrier(carrier);
  try {
    assertTransactionInputDepth(input, depth);
  } catch (error) {
    if (!(error instanceof TransactionDevelopmentLimitError)) throw error;
    depthRefusals.set(result, error.message);
  }
  return result;
}
/** Source-private, read-only fact; only this module can create or annotate its brand. */
export function transactionDepthRefusal(batch: PreparedTransaction): string | undefined {
  if (!prepared.has(batch)) throw new TypeError("expected prepared transaction");
  return depthRefusals.get(batch);
}
function requireTransactionDepth(batch: PreparedTransaction): PreparedTransaction {
  const message = transactionDepthRefusal(batch);
  if (message !== undefined) throw new TransactionDevelopmentLimitError(message);
  return batch;
}
export function assertPreparedTransaction(batch: PreparedTransaction, scope: string): void {
  if (!prepared.has(batch) || batch.carrier.scope !== scope)
    throw new TypeError("expected prepared transaction for this Scope");
}
/** Source-private permitted alias slots shared by preparation and pending validation. */
export function transactionAliasFields(operation: string): readonly string[] {
  if (operation === "createLink") return ["source", "target"];
  if (operation === "updateBead" || operation === "deleteBead") return ["bead"];
  return [];
}

## Parser container freeze and complete JSON decoder
export function decodeJsonDocument(text: string): LosslessJsonValue {
  if (typeof text !== "string") throw new TypeError("JSON input must be text");
  let at = 0;
  const frames: Frame[] = [];
  const whitespace = (): void => {
    while (at < text.length && /[\x20\t\r\n]/.test(text[at] ?? "")) at++;
  };
  const fail = (message: string): never => {
    throw new JsonSyntaxError(message, at);
  };
  const string = (): string => {
    const start = at++;
    let escaped = false;
    while (at < text.length) {
      const char = text[at++];
      if (char === '"' && !escaped) {
        let value: string;
        try {
          value = JSON.parse(text.slice(start, at)) as string;
        } catch {
          return fail("invalid JSON string");
        }
        if (!isUnicodeScalarString(value)) fail("unpaired surrogate");
        return value;
      }
      if (char === "\\" && !escaped) escaped = true;
      else escaped = false;
    }
    return fail("unterminated JSON string");
  };
  const value = (): LosslessJsonValue => {
    whitespace();
    const char = text[at];
    if (char === '"') return string();
    if (char === "{" || char === "[") {
      at++;
      const object = char === "{";
      const result: MutableContainer = object
        ? (Object.create(null) as Record<string, LosslessJsonValue>)
        : [];
      frames.push({ value: result, object, state: "first", keys: new Set() });
      return result;
    }
    for (const [token, literal] of [
      ["true", true],
      ["false", false],
      ["null", null],
    ] as const) {
      if (text.startsWith(token, at)) {
        at += token.length;
        return literal;
      }
    }
    NUMBER_PREFIX.lastIndex = at;
    const matched = NUMBER_PREFIX.exec(text);
    if (matched) {
      at = NUMBER_PREFIX.lastIndex;
      return new JsonNumberLiteral(matched[0]);
    }
    return fail("expected JSON value");
  };
  const root = value();
  while (frames.length) {
    const frame = frames[frames.length - 1];
    if (!frame) throw new Error("missing JSON frame");
    whitespace();
    const close = frame.object ? "}" : "]";
    if (text[at] === close && frame.state !== "next") {
      at++;
      Object.freeze(frame.value);
      frames.pop();
      continue;
    }
    if (frame.state === "after") {
      if (text[at++] !== ",") fail("expected comma or closing delimiter");
      frame.state = "next";
      continue;
    }
    let key: string | undefined;
    if (frame.object) {
      if (text[at] !== '"') fail("expected object member name");
      key = string();
      if (frame.keys.has(key)) fail("duplicate decoded object member name");
      frame.keys.add(key);
      whitespace();
      if (text[at++] !== ":") fail("expected colon");
    }
    frame.state = "after";
    const child = value();
    if (Array.isArray(frame.value)) frame.value.push(child);
    else if (key !== undefined) frame.value[key] = child;
  }
  whitespace();
  if (at !== text.length) fail("trailing JSON input");
  return root;
}

## Operation freeze and carrier parser
const parsedOperations = new WeakSet<object>();
function operation<K extends ReadUpdateOperation>(
  kind: K,
  input: Readonly<Record<string, LosslessJsonValue>>,
  key?: string,
  name?: string,
): UnadmittedReadUpdateOperation<K> {
  const value = Object.freeze({
    operation: kind,
    input,
    ...(key === undefined ? {} : { idempotencyKey: key }),
    ...(name === undefined ? {} : { name }),
  });
  parsedOperations.add(value);
  return value;
}
function record(value: LosslessJsonValue): Readonly<Record<string, LosslessJsonValue>> {
  if (!value || typeof value !== "object" || Array.isArray(value))
    throw new ReadUpdateCarrierError("operation record must be an object");
  return value as Readonly<Record<string, LosslessJsonValue>>;
}
function carrier(
  text: string,
  key: keyof typeof READ_UPDATE_VALUE_SCHEMA_REFS,
): Readonly<Record<string, LosslessJsonValue>> {
  const root = decodeJsonDocument(text);
  // Numbers are only structural placeholders here, private and discarded.
  // The reachable-root invariant in read-update-numeric-schema.test.ts guards
  // that placeholders cannot turn valid property literals into syntax errors.
  const structural = mapLosslessJson(root, () => 0);
  // Schema loading/validator failures are internal faults, never blamed on input.
  const check = validator(key);
  if (!check(structural)) throw new ReadUpdateCarrierError(`Malformed ${key} carrier`);
  return record(root);
}

## Complete sequence syntax and operation-input capture
export function parseReadUpdateSequenceRequest(text: string): ReadUpdateSequenceRequest {
  const root = carrier(text, "sequenceRequest");
  const operations = root.operations as readonly LosslessJsonValue[];
  const keys = new Set<string>();
  const names = new Map<string, "bead" | "link">();
  const parsed: UnadmittedReadUpdateOperation[] = [];
  for (const member of operations) {
    const fields = record(member);
    const kind = fields.operation as ReadUpdateOperation;
    validateReferenceSyntax(kind, fields);
    validateAliasSyntax(kind, fields);
    const key = fields.idempotencyKey as string;
    const name = fields.name as string | undefined;
    if (keys.has(key)) throw new ReadUpdateCarrierError("duplicate sequence idempotencyKey");
    keys.add(key);
    const expected: Readonly<Record<string, "bead" | "link">> =
      kind === "createLink"
        ? { source: "bead", target: "bead" }
        : kind === "putAlias"
          ? { target: "bead" }
          : kind === "updateBead" || kind === "deleteBead"
            ? { bead: "bead" }
            : kind === "updateLink" || kind === "deleteLink"
              ? { link: "link" }
              : {};
    for (const [field, resourceKind] of Object.entries(expected)) {
      const reference = fields[field];
      const uri = typeof reference === "string" ? reference : record(reference ?? null).uri;
      if (
        typeof uri === "string" &&
        uri.startsWith("@") &&
        names.get(uri.slice(1)) !== resourceKind
      )
        throw new ReadUpdateCarrierError("forward, unknown or wrong-kind sequence binding");
    }
    // Register only after references: a creator cannot reference its own name.
    if (name !== undefined) {
      if (names.has(name)) throw new ReadUpdateCarrierError("duplicate sequence name");
      names.set(name, kind === "createBead" ? "bead" : "link");
    }
    const input = Object.freeze(
      Object.fromEntries(
        Object.entries(fields).filter(
          ([key]) => key !== "operation" && key !== "idempotencyKey" && key !== "name",
        ),
      ),
    );
    parsed.push(operation(kind, input, key, name));
  }
  return Object.freeze({ operations: Object.freeze(parsed) });

## Complete installed launcher lifecycle
/** Bounded inherited-IPC launcher. No request or observation emulation. */
import { spawn } from "node:child_process";
import { mkdir } from "node:fs/promises";
export async function launchInstalledTransaction(entry, configuration, signal) {
  signal?.throwIfAborted();
  // Explicit launcher-owned provisioning; reopen keeps the directory/create:false.
  if (configuration.create) await mkdir(configuration.directory, { recursive: true, mode: 0o700 });
  signal?.throwIfAborted();
  const child = spawn(process.execPath, [entry], {
    cwd: "/tmp",
    stdio: ["ignore", "pipe", "pipe", "ipc"],
  });
  let bytes = 0;
  for (const stream of [child.stdout, child.stderr])
    stream.on("data", (chunk) => {
      bytes += chunk.length;
      if (bytes > 65536) child.kill("SIGKILL");
    });
  const exited = new Promise((resolve) =>
    child.once("exit", (code, signal) => resolve({ code, signal })),
  );
  async function bounded(promise) {
    let timer;
    try {
      return await Promise.race([
        promise,
        new Promise((_, reject) => {
          timer = setTimeout(() => {
            child.kill("SIGKILL");
            reject(new Error("installed transaction process deadline"));
          }, 10000);
        }),
      ]);
    } finally {
      clearTimeout(timer);
    }
  }
  async function drainFailure(error) {
    child.kill("SIGKILL");
    try {
      await bounded(exited);
    } catch (cleanup) {
      throw new AggregateError([error, cleanup], "installed transaction cleanup failed");
    }
    throw error;
  }
  let closing;
  const close = () =>
    (closing ??= (async () => {
      try {
        if (child.connected) child.send({ command: "close" });
        const result = await bounded(exited);
        if (result.code !== 0 || result.signal !== null)
          throw new Error("installed transaction child did not close cleanly");
      } catch (error) {
        await drainFailure(error);
      } finally {
        signal?.removeEventListener("abort", onAbort);
      }
    })());
  const onAbort = () => {
    void close().catch(() => {});
  };
  signal?.addEventListener("abort", onAbort, { once: true });
  if (signal?.aborted) onAbort();
  let detachReadyAbort = () => {};
  try {
    const ready = await bounded(
      new Promise((resolve, reject) => {
        const abort = () => reject(signal.reason);
        signal?.addEventListener("abort", abort, { once: true });
        detachReadyAbort = () => signal?.removeEventListener("abort", abort);
        if (signal?.aborted) {
          abort();
          return;
        }
        child.once("error", reject);
        child.once("message", resolve);
        void exited.then(() => reject(new Error("installed transaction exited before ready")));
        child.send({ configuration });
      }),
    );
    signal?.throwIfAborted();
    if (!ready || ready.event !== "qualification.ready" || JSON.stringify(ready).length > 65536)
      throw new Error("invalid installed ready message");
    return { child, ready, close };
  } catch (error) {
    try {
      await close();
    } catch (cleanup) {
      throw new AggregateError([error, cleanup], "installed transaction startup cleanup failed");
    }
    throw error;
  } finally {
    detachReadyAbort();
  }
}
