19:         vary !== "*" &&
20:         !vary?.split(",").some((field) => field.trim().toLowerCase() === "accept")
21:       )
22:         headers.set("vary", vary ? `${vary}, Accept` : "Accept");
23:     }
24:     // The Scope probe is an existing zero-length representation with no media
25:     // type to negotiate; it still evaluates entity-tag preconditions.
26:     const status =

29:         : conditionalReadStatus(request.headers, headers.get("etag"));
