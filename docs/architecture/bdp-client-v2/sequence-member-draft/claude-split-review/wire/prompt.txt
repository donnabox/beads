export function receiptUrl(value: unknown, scope: string): string {
  if (typeof value !== "string" || value.length > 8192)
    throw new TransactionalHarnessError("unobserved receipt target");
  let url: URL;
  try {
    url = new URL(value);
  } catch (cause) {
    throw new TransactionalHarnessError("unobserved receipt target", { cause });
  }
  const prefix = `${scope}receipts/`;
  if (
    url.href !== value ||
    !value.startsWith(prefix) ||
    !/^[A-Za-z0-9_-]+$/.test(value.slice(prefix.length)) ||
    url.search ||
    url.hash ||
    url.username ||
    url.password

/** This bounded fixture requires the receiver's actual Content-Length encoding.
 * No general alternate transfer coding claim is made by this witness. */
export function validateReceiptWire(response: HttpExchangeResponse, method: "GET" | "HEAD") {
  const bytes = response.wireResponseBytes;
  if (!(bytes instanceof Uint8Array) || bytes.byteLength > 2097152)
    throw new TransactionalHarnessError("receipt wire evidence unavailable");
  const wire = Buffer.from(bytes),
    delimiter = wire.indexOf("\r\n\r\n");
  if (delimiter < 0 || delimiter > 65536)
    throw new TransactionalHarnessError("receipt wire header bound");
  const lines = wire.subarray(0, delimiter).toString("latin1").split("\r\n");
  if (lines.shift() !== `HTTP/1.1 ${response.status} OK`)
    throw new TransactionalHarnessError("receipt wire status mismatch");
  const actual: Record<string, string> = {};
  for (const line of lines) {
    const colon = line.indexOf(":");
    if (colon < 1) throw new TransactionalHarnessError("receipt wire header mismatch");
    const name = line.slice(0, colon).toLowerCase(),
      value = line.slice(colon + 1).trim();
    if (Object.hasOwn(actual, name))
      throw new TransactionalHarnessError("duplicate receipt header");
    actual[name] = value;
  }
  const expected = Object.fromEntries(
    Object.entries(response.headers).map(([k, v]) => [k.toLowerCase(), v]),
  );
  if (!isDeepStrictEqual(actual, expected))
    throw new TransactionalHarnessError("receipt wire headers mismatch");
  const body = wire.subarray(delimiter + 4);
  if (
    body.length > 1048576 ||
    response.bodyOctets !== body.length ||
    !body.equals(Buffer.from(response.bodyText)) ||
    (method === "HEAD" && body.length !== 0) ||
    (method === "GET" && Number(actual["content-length"]) !== body.length)
  )
    throw new TransactionalHarnessError("receipt wire body mismatch");
}

function project(sequence: PreparedTransactionSequence, turn: TransactionSequenceTurn): Entry {
  const member = sequence.members[turn.index];
  if (!member) throw new Error("sequence turn outside carrier");
  let entry: Record<string, unknown>;
  if (turn.result.kind === "transient") entry = { ...problem("idempotency-in-progress") };
  else if (turn.result.kind === "conflict") entry = { ...problem("idempotency-conflict") };
  else {
    const receipt = turn.result.receipt;
    if (receipt.status === "pending") entry = { ...problem("idempotency-in-progress") };
    else if (receipt.detail === "expired") {
      if (receipt.status !== "completed") throw new Error("expired noncompleted receipt");
      entry = { ...problem("idempotency-expired") };
      if (member.operation === "createBead" || member.operation === "createLink") {
        if (!Array.isArray(receipt.allocated) || receipt.allocated.length !== 1)
          throw new Error("expired singleton creation lost its allocation");
        const allocated = record(receipt.allocated[0]);
        if (
          allocated.operationIndex !== 0 ||
          typeof allocated.id !== "string" ||
          typeof allocated.type !== "string"
        )
          throw new Error("invalid expired singleton allocation");
        entry.allocated = { id: allocated.id, type: allocated.type };
      }
    } else if (receipt.detail !== "available")
      throw new Error("unsupported private receipt disclosure");
    else if (receipt.status === "failed") entry = { ...record(receipt.problem) };
    else {
      if (!Array.isArray(receipt.results) || receipt.results.length !== 1 || receipt.next !== null)
        throw new Error("singleton sequence receipt must have exactly one result");
      entry = { ...record(receipt.results[0]) };
    }
  }
  // The current presentation supplies the position and label. Retained result
  // labels belong to the original carrier and are not semantic identity.
  delete entry.operationIndex;
  delete entry.operationName;
  return Object.freeze({
    ...entry,
    operationIndex: turn.index,
    ...(member.name === undefined ? {} : { operationName: member.name }),
  });
}

  source: string,

  };

export function projectTransactionEventGroup(
  value: unknown,
  context: TransactionReadContext,
  source: TransactionEventSource,
  scope: string,
  expectedPosition: number,
): readonly TransactionEvent[] {
  if (!value || typeof value !== "object") throw new Error("invalid retained Event group");
  const group = value as TransactionEventGroup;
  if (
    group.scopeEpoch !== context.scopeEpoch ||
    group.authorizationView !== context.authorizationView ||
    group.position !== `pos-${expectedPosition}` ||
    group.checkpoint !== group.position ||
    !Array.isArray(group.events) ||
    group.eventCount !== group.events.length ||
    !Array.isArray(group.changes) ||
    typeof group.transaction !== "string"
  )
    throw new Error("retained Event group context mismatch");
  const endpoints = new Map<string, readonly [unknown, unknown]>();
  const linkIds = new Set<string>();
  const remember = (id: string, from: unknown, to: unknown) => {
    reference(from);
    reference(to);
    const old = endpoints.get(id),
      next = [from, to] as const;
    if (old && stringifyJsonValue(old) !== stringifyJsonValue(next))
      throw new Error("inconsistent retained Link endpoints");
    endpoints.set(id, next);
  };
  for (const change of group.changes) {
    if (change.resourceKind === "link") {
      linkIds.add(change.resource.id);
      if (change.operation === "upsert" && "source" in change.resource)
        remember(change.resource.id, change.resource.source, change.resource.target);
    }
  }
  for (const event of group.events) {
    const data = event.data as Record<string, unknown>;
    if (!data || typeof data !== "object") throw new Error("invalid retained Event data");
    if (
      event.type === "linked" ||
      event.type === "unlinked" ||
      (event.type === "created" && linkIds.has(event.subject))
    )
      remember(event.subject, data.source, data.target);
  }
  const result: TransactionEvent[] = [];
  for (const [index, event] of group.events.entries()) {
    if (
      event.ordinal !== index ||
      event.source !== `${scope}events/` ||
      event.transaction !== group.transaction ||
      !["created", "updated", "deleted", "linked", "unlinked"].includes(event.type)
    )
      throw new Error("invalid retained Event framing");
    const data = event.data as Record<string, unknown>;
    let include = source.kind === "scope";
    if (source.kind !== "scope") {
      include =
        event.subject === source.subject && ["created", "updated", "deleted"].includes(event.type);
      if (source.kind === "bead" && (event.type === "linked" || event.type === "unlinked")) {
        if (data.endpoint !== "source" && data.endpoint !== "target")
          throw new Error("invalid endpoint fact");
        include = reference(data[data.endpoint]) === source.subject;
      }
      if (source.kind === "bead" && event.type === "updated" && linkIds.has(event.subject)) {
        const pair = endpoints.get(event.subject);
        if (!pair) throw new Error("incident Link update lacks same-group endpoints");
        include = pair.some((endpoint) => reference(endpoint) === source.subject);
        // Literal incident rule currently mentions properties only. Do not silently
        // decide metadata-only membership while the spec disposition is pending.
        if (include && data.change === undefined && data.metadataChange !== undefined)
          throw new TransactionEventProjectionGap("incident-link-metadata");
      }
    }
    if (include) result.push(renderTransactionEvent(context, source.url, group.checkpoint, event));
  }
  return result;
}
