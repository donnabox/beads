The frozen successor incorrectly says existing execution deadlines bound every drain. The actual private service scheduler explicitly provides no delivery timeout. Root accepted this correction before runtime execution.

The trusted service/runtime replacement requires timeoutMs and optional signal, with timeoutMs a positive safe integer at most 2147483647. The service owns and clears its timer and abort listener. Every delivery of a receipt — the synchronous response, the response to a
duplicate, a later `GET` or `HEAD`, every page, and the projection of a
member's receipt into a sequence response — is one representation, the
receipt as retained projected under the serving request's current
Authorization View. An entry that carries a Resource record — `created`
or `updated` — is served only when the current view projects that record
as retained, whether or not the Resource still exists or is at that
revision, and the view's closure over owned Links applies: hiding a Bead
hides the entries of every Bead that owns a Link to it and of those Links.
An entry the view does not project is served as `withheld`, carrying only
`operationIndex` and, when present, `operationName`; an `allocated` entry
is re-authorized the same way and carries `withheld` `true` in place of its
identity. Entries that carry no record — `matched` counts, `deleted` and
`erased` identities, and `withheld` entries — and a `failed` receipt's
`problem` are the transaction's own execution facts, disclosed to the
principal when it executed, and are served as retained. Alias result entries
are non-Resource execution facts under this same rule.
When a `completed`
receipt's every entry is withheld, the receipt carries `detail` `withheld`
and neither `results`, `next`, nor `allocated`; a `failed` receipt is never
`withheld`. The disposition, `requiredPosition`, and `effectPosition` are
never withheld. Withholding is authorization, never deletion: an entry
whose Resource was since deleted is served like any other when the view
projects its retained record.

export function parseTransactionAuthorization(text: string): TransactionAuthorizationSnapshot {
  if (Buffer.byteLength(text) > 4096) throw new Error("authorization policy exceeds storage bound");
  const v: unknown = JSON.parse(text);
  if (
    !v ||
    typeof v !== "object" ||
    Array.isArray(v) ||
    Object.keys(v).sort().join(",") !==
      "authorizationView,bindingKind,canMutate,generation,readPolicy,revision"
  )
    throw new Error("invalid authorization policy fields");
  const p = v as TransactionAuthorizationSnapshot;
  if (
    p.revision !== 1 ||
    p.bindingKind !== "scope" ||
    !(
      (p.generation === 0 &&
        p.authorizationView === INITIAL_TRANSACTION_VIEW &&
        p.readPolicy === "all-visible" &&
        p.canMutate === true) ||
      (p.generation === 1 &&
        typeof p.authorizationView === "string" &&
        /^[0-9a-f]{8}-[0-9a-f]{4}-4[0-9a-f]{3}-[89ab][0-9a-f]{3}-[0-9a-f]{12}$/.test(

export function projectTransactionEntry(value: unknown, visible: boolean): unknown {
  if (visible || !value || typeof value !== "object" || Array.isArray(value)) return value;

    return value;
  return {
    operationIndex: entry.operationIndex,
    ...(entry.operationName === undefined ? {} : { operationName: entry.operationName }),
    outcome: "withheld",
  };
}

interface Sink {
  readonly destroyed: boolean;
  write(value: string): boolean;
  once(event: "drain" | "close", listener: () => void): unknown;
  once(event: "error", listener: (error: Error) => void): unknown;

/** write(false) accepted this frame into Node, but permits no subsequent owner
 * read/comment until drain. It is not a client replica acknowledgement. */
async function write(sink: Sink, value: string, signal: AbortSignal): Promise<void> {
  signal.throwIfAborted();
  if (sink.destroyed) throw new Error("stream transport closed");
  if (sink.write(value)) return;
  await new Promise<void>((resolve, reject) => {
    const clean = () => {
      sink.removeListener("drain", drained);
      sink.removeListener("error", failed);
      sink.removeListener("close", closed);
      signal.removeEventListener("abort", aborted);
    };
    const drained = () => {
      clean();
      resolve();
    };
    const failed = (error: unknown) => {
      clean();
      reject(error);
    };
    const closed = () => failed(new Error("stream transport closed"));
    const aborted = () => failed(signal.reason);
    sink.once("drain", drained);
    sink.once("error", failed);
    sink.once("close", closed);
    signal.addEventListener("abort", aborted, { once: true });
    if (signal.aborted) aborted();
    else if (sink.destroyed) closed();
  });
}

interface Packet<Item, Cursor> {
  readonly items: readonly Item[];
  readonly continuation: Cursor;
  readonly caughtUp: boolean;
}
type Timers = { readonly pollMilliseconds: number; readonly heartbeatMilliseconds: number };
/** One private delivery loop. It retains only the current packet; continuations
 * advance after its final drain, never on a speculative owner pull. */
async function pump<Item, Cursor>(
  sink: Sink,
  initial: Packet<Item, Cursor> | undefined,
  next: (cursor: Cursor, signal: AbortSignal) => Promise<Packet<Item, Cursor>>,
  frame: (item: Item) => string,
  signal: AbortSignal,
  settings: Timers,
): Promise<void> {
  validateTransactionStreamTimers(settings);
  let packet = initial;
  initial = undefined;
  let lastWrite = performance.now();
  while (packet && !signal.aborted) {
    for (const item of packet.items) {
      await write(sink, frame(item), signal);
      lastWrite = performance.now();
    }
    const continuation = packet.continuation,
      caughtUp = packet.caughtUp;
    packet = undefined;
    // Empty historical packets must keep this check active too. Comments obey
    // the same write/drain barrier; no timer queues independent writes.
    if (performance.now() - lastWrite >= settings.heartbeatMilliseconds) {
      await write(sink, ": heartbeat\n\n", signal);
      lastWrite = performance.now();
    }
    if (caughtUp) await delay(settings.pollMilliseconds, undefined, { signal });
    else await yieldTurn(undefined, { signal });
    signal.throwIfAborted();
    packet = await next(continuation, signal);
  }
}
