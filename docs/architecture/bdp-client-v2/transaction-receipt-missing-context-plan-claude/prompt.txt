export async function waitForTransactionMinimum<T>(
  attempt: () => T | Promise<T>,
  options: TransactionMinimumWaitOptions = {},
  signal?: AbortSignal,
): Promise<T> {
  const budget = options.budgetMs ?? 1000,
    now = options.now ?? (() => performance.now()),
    wait = options.wait ?? ((milliseconds) => delay(milliseconds));
  if (
    !Number.isSafeInteger(budget) ||
    budget <= 0 ||
    budget > 2147483647 ||
    typeof now !== "function" ||
    typeof wait !== "function"
  )
    throw new TypeError("finite minimum-read wait policy required");
  const started = now();
  if (!Number.isFinite(started)) throw new TypeError("invalid minimum-read clock");
  const deadline = started + budget;
  let previous = started;
  if (!Number.isFinite(deadline)) throw new TypeError("invalid minimum-read deadline");
  for (;;) {
    if (signal?.aborted) throw new ScopeServerOperationAbortedError();
    try {
      return await attempt();
    } catch (error) {
      if (!(error instanceof TransactionMinimumFailure) || error.reason !== "behind") throw error;
      const current = now();
      if (!Number.isFinite(current) || current < previous)
        throw new TypeError("invalid minimum-read clock");
      previous = current;
      if (current >= deadline)
        throw new TransactionMinimumFailure("catch-up-timeout", error.context);
      await wait(Math.min(5, deadline - current));
    }
  }
}

/** Bounded inherited-IPC launcher. No request or observation emulation. */
import { spawn } from "node:child_process";
import { mkdir } from "node:fs/promises";
export async function launchInstalledTransaction(entry, configuration, signal) {
  signal?.throwIfAborted();
  // Explicit launcher-owned provisioning; reopen keeps the directory/create:false.
  if (configuration.create) await mkdir(configuration.directory, { recursive: true, mode: 0o700 });
  signal?.throwIfAborted();
  const child = spawn(process.execPath, [entry], {
    cwd: "/tmp",
    stdio: ["ignore", "pipe", "pipe", "ipc"],
  });
  let bytes = 0;
  for (const stream of [child.stdout, child.stderr])
    stream.on("data", (chunk) => {
      bytes += chunk.length;
      if (bytes > 65536) child.kill("SIGKILL");
    });
  const exited = new Promise((resolve) =>
    child.once("exit", (code, signal) => resolve({ code, signal })),
  );
  async function bounded(promise) {
    let timer;
    try {
      return await Promise.race([
        promise,
        new Promise((_, reject) => {
          timer = setTimeout(() => {
            child.kill("SIGKILL");
            reject(new Error("installed transaction process deadline"));
          }, 10000);
        }),
      ]);
    } finally {
      clearTimeout(timer);
    }
  }
  async function drainFailure(error) {
    child.kill("SIGKILL");
    try {
      await bounded(exited);
    } catch (cleanup) {
      throw new AggregateError([error, cleanup], "installed transaction cleanup failed");
    }
    throw error;
  }
  let closing;
  const close = () =>
    (closing ??= (async () => {
      try {
        if (child.connected) child.send({ command: "close" });
        const result = await bounded(exited);
        if (result.code !== 0 || result.signal !== null)
          throw new Error("installed transaction child did not close cleanly");
      } catch (error) {
        await drainFailure(error);
      } finally {
        signal?.removeEventListener("abort", onAbort);
      }
    })());
  const onAbort = () => {
    void close().catch(() => {});
  };
  signal?.addEventListener("abort", onAbort, { once: true });
  if (signal?.aborted) onAbort();
  let detachReadyAbort = () => {};
  try {
    const ready = await bounded(
      new Promise((resolve, reject) => {
        const abort = () => reject(signal.reason);
        signal?.addEventListener("abort", abort, { once: true });
        detachReadyAbort = () => signal?.removeEventListener("abort", abort);
        if (signal?.aborted) {
          abort();
          return;
        }
        child.once("error", reject);
        child.once("message", resolve);
        void exited.then(() => reject(new Error("installed transaction exited before ready")));
        child.send({ configuration });
      }),
    );
    signal?.throwIfAborted();
    if (!ready || ready.event !== "qualification.ready" || JSON.stringify(ready).length > 65536)
      throw new Error("invalid installed ready message");
    return { child, ready, close };
  } catch (error) {
    try {
      await close();
    } catch (cleanup) {
      throw new AggregateError([error, cleanup], "installed transaction startup cleanup failed");
    }
    throw error;
  } finally {
    detachReadyAbort();
  }
}

### docs/specs/bdp.md:5116-5133
```
Every admitted mutation has one Mutation Receipt at an authority-allocated
URL beneath the discovered `receipts` root — `receipts/{token}`, where the
token is a checkpoint-profile token — and the receipt's `id` is that
absolute URL. `GET` and `HEAD` of the receipt URL return the receipt's
current representation with `200 OK`, `Cache-Control: private, no-store`,
and the Transactional response fields. A receipt URL that does not exist,
that belongs to another principal, that was retracted under
[Mutation Transactions](#mutation-transactions), that named a failed
receipt the authority has since forgotten under the same section, or that
was allocated in another Scope epoch returns the uniform `404`
`resource-not-found`: receipts are principal-bound, and they are not an
enumeration oracle. The `receipts` root is a namespace prefix, not a
Resource: it is discovered so that receipt URLs are recognizably the
authority's and clients never construct them, BDP v0 defines no receipt
listing and no lookup by key, and a `GET` of the root returns `404`
`resource-not-found`. A client resolves a lost response by retrying the
original request with its original key, which returns the receipt.
